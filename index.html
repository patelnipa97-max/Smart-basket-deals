import requests
from bs4 import BeautifulSoup
import json
from datetime import datetime

# ==========================================
# SmartBasket & HeartyMeals - Weekly Scraper
# Target Stores: Whole Foods, Woodmans, JewelOsco, Target, Patel Brothers
# ==========================================

STORES_CONFIG = {
    "Whole Foods": {"url": "https://www.wholefoodsmarket.com/products/sale", "item_selector": "div.aliment-product-card"},
    "Woodmans": {"url": "https://www.woodmans-food.com/weekly-ad", "item_selector": "div.circular-item"},
    "JewelOsco": {"url": "https://local.jewelosco.com/weekly-ad", "item_selector": "div.flyer-item"},
    "Target": {"url": "https://www.target.com/c/weekly-ad", "item_selector": "div.h-padding-a-default"},
    "Patel Brothers": {"url": "https://www.patelbros.com/weekly-ad", "item_selector": "div.product-tile"}
}

def fetch_store_circulars():
    headers = {"User-Agent": "SmartBasketBot/1.0 (Contact: nipa@smartbasket.app)"}
    scraped_deals = []

    print(f"[{datetime.now()}] Starting weekly grocery circular sweep across 5 stores...")

    for store_name, config in STORES_CONFIG.items():
        try:
            response = requests.get(config["url"], headers=headers, timeout=10)
            if response.status_code == 200:
                soup = BeautifulSoup(response.text, 'html.parser')
                items = soup.select(config["item_selector"])
                
                for item in items[:3]: # Sample top 3 weekly deals per store
                    title = item.select_one('.title, h3, span.name').get_text(strip=True) if item.select_one('.title, h3, span.name') else "Organic Special Item"
                    price = item.select_one('.price, span.sale-price').get_text(strip=True) if item.select_one('.price, span.sale-price') else "$3.99"
                    
                    scraped_deals.append({
                        "store": store_name,
                        "name": title,
                        "salePrice": price,
                        "timestamp": datetime.now().isoformat()
                    })
            else:
                print(f"Warning: Could not reach {store_name} (Status: {response.status_code})")
        except Exception as e:
            print(f"Error scraping {store_name}: {str(e)}")

    print(f"Successfully scraped {len(scraped_deals)} active deals into JSON format.")
    return json.dumps(scraped_deals, indent=2)

if __name__ == "__main__":
    deals_json = fetch_store_circulars()
    with open("live_weekly_deals.json", "w") as f:
        f.write(deals_json)
    print("Weekly circular sync completed successfully!")

The Inspiration

As a mom to a wonderful daughter, I often look back at my early years watching my parents and family members meticulously collect coupons and hunt for deals. They showed me firsthand how much love, care, and smart planning go into saving a few dollars on household groceries and everyday necessities.

The Modern Struggle

When I started my own household, I quickly realized how difficult it is to keep that tradition alive in today's busy world. Trying to gather paper coupons, track down digital deals, and drive across town to multiple different grocery stores—all while juggling a demanding career, weekend schedules, and the desire to feed my family fresh, healthy meals—felt nearly impossible.

The IT Solution: SmartBasket & HeartyMeals

That’s when my background in IT and data analytics kicked in. I wanted to build something that would make life easier for busy parents and coupon-lovers alike. SmartBasket & HeartyMeals was created to be your ultimate one-stop shop—bringing grocery deals, weekly circulars, and healthy, dietary-friendly meal planning from stores like Whole Foods, Woodman's, Jewel-Osco, Target, and Patel Brothers into one easy-to-access place.
